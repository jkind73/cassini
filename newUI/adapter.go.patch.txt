====================

adapter.go. Rationale: DiscInfo already reads sector 0's IP System ID

	ipGameTitle   = 0x60 // 112 bytes, space padded

+// parseAreaCode parses the IP System ID's Compatible Area Symbols
+// field into the single best-match Region for BIOS auto-assignment.
+// ip_bin.md: "J=Japan, T=Asia NTSC, U=North America, E=Europe PAL",
+// space padded, multiple symbols possible (e.g. "JTU " for a
+// tri-region disc). Preference order below matches which BIOS region
+// a disc most needs to boot correctly when it declares more than
+// one: a disc listing J needs a Japan-region BIOS's security check
+// to pass, ahead of also being playable on overseas hardware.
+func parseAreaCode(field string) uiface.Region {
+	s := strings.TrimSpace(field)
+	switch {
+	case strings.Contains(s, "J"):
+		return uiface.RegionJapan
+	case strings.Contains(s, "U"):
+		return uiface.RegionUSA
+	case strings.Contains(s, "E"):
+		return uiface.RegionEurope
+	case strings.Contains(s, "T"):
+		return uiface.RegionAsiaNTSC
+	default:
+		return uiface.RegionUnknown
+	}
+}
